# ============================================
# Publication-quality figure with 600 DPI
# ============================================
import matplotlib.pyplot as plt
import matplotlib as mpl

# --- تنظیمات استاندارد IEEE ---
mpl.rcParams.update({
    'font.family': 'serif',
    'font.serif': ['Times New Roman'],
    'font.size': 10,
    'axes.labelsize': 10,
    'axes.titlesize': 11,
    'legend.fontsize': 9,
    'figure.dpi': 600,
    'savefig.dpi': 600,
    'savefig.bbox': 'tight',
    'savefig.format': 'png',
    'axes.grid': True,
    'grid.alpha': 0.3,
})

# --- داده ---
import numpy as np
x = np.linspace(0، 1، 100)
u_exact = np.sin(np.pi * x)
u_pinn = u_exact + 0.01 * np.random.randn(100)

# --- نمودار ---
fig، ax = plt.subplots(figsize=(3.5، 2.5))  # single column
ax.plot(x، u_exact، 'k-'، lw=1.5، label='Exact')
ax.plot(x، u_pinn، 'r--'، lw=1.5، label='PINN')
ax.set_xlabel('Position x')
ax.set_ylabel('Temperature u')
ax.set_title('Heat Equation Solution')
ax.legend(loc='best')

# --- ذخیره با کیفیت ۶۰۰ DPI ---
plt.savefig('figure1.png'، dpi=600، bbox_inches='tight')
plt.savefig('figure1.pdf'، bbox_inches='tight')  # vector format
plt.show()

# --- نتیجه: تصویر آماده برای IEEE ---
print('✅ Figure saved at 600 DPI')

زندگی و مقام علمی

آیت‌الله حسن حسن‌زاده آملی (۱۳۰۷–۱۴۰۰ ش) از برجسته‌ترین دانشمندان معاصر ایران بود که در علوم مختلف تبحر داشتند:

  • فقه و اصول — از شاگردان برجسته علامه طباطبایی
  • فلسفه — حکمت متعالیه و فلسفه اسلامی
  • عرفان نظری و عملی — از عرفای بزرگ معاصر
  • ریاضیات و نجوم — حدود ۱۷ سال تدریس هیئت و ریاضیات
  • ادبیات — عربی و فارسی
تألیفات

ایشان حدود ۱۹۰ اثر در موضوعات مختلف تألیف کردند، از جمله:

  • «معرفة الوقت و القبلة» — درس‌های ریاضیات و نجوم
  • «شرح اشارات» — فلسفه ابن سینا
  • «اتحاد عاقل و معقول» — فلسفه و معرفت‌شناسی
  • «الهی‌نامه» — عرفان
پیوند با جلسه ۲۴

آیت‌الله حسن‌زاده آملی نماد ترکیب علم و معنویت و ذوفنون بودن هستند. در جلسه پایانی دوره، از ایشان درس می‌گیریم که:

  • علم، محدود به یک رشته نیست
  • دقت علمی و اخلاق پژوهش، همراه هم هستند
  • ریاضیات، زبان فهم جهان است
  • پژوهش، عبادتی علمی است
درس امروز: همان‌طور که آیت‌الله حسن‌زاده آملی در علوم مختلف تبحر داشتند، شما نیز در این دوره از ریاضیات تا ML سفر کردید. علم، یک کل به‌هم‌پیوسته است.
